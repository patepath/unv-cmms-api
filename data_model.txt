// 1. Building Model (ตารางอาคาร / สถานที่)
type Building struct {
	BuildingCode string      `gorm:"primaryKey;size:20;column:building_code" json:"building_code"`
	BuildingName string      `gorm:"size:255;not null;column:building_name" json:"building_name"`
	CreatedAt    time.Time   `gorm:"autoCreateTime;column:created_at" json:"created_at"`
	Equipments   []Equipment `gorm:"foreignKey:BuildingCode;references:BuildingCode" json:"equipments,omitempty"`
}

func (Building) TableName() string {
	return "buildings"
}

// 2. EquipmentCategory Model (ตารางหมวดหมู่/ประเภทอุปกรณ์)
type EquipmentCategory struct {
	CategoryID    string      `gorm:"primaryKey;size:30;column:category_id" json:"category_id"`
	CategoryName  string      `gorm:"size:100;not null;column:category_name" json:"category_name"`
	CategoryGroup string      `gorm:"size:50;column:category_group" json:"category_group"`
	Equipments    []Equipment `gorm:"foreignKey:CategoryID;references:CategoryID" json:"equipments,omitempty"`
}

func (EquipmentCategory) TableName() string {
	return "equipment_categories"
}

// 3. Equipment Model (ตารางอุปกรณ์หลัก)
type Equipment struct {
	EquipmentID          string   `gorm:"primaryKey;size:50;column:equipment_id" json:"equipment_id"`
	CategoryID           string   `gorm:"size:30;not null;index;column:category_id" json:"category_id"`
	BuildingCode         *string  `gorm:"size:20;index;column:building_code" json:"building_code,omitempty"`
	InstallationLocation string   `gorm:"size:255;column:installation_location" json:"installation_location"`
	Floor                string   `gorm:"size:20;column:floor" json:"floor"`
	AssetNo              string   `gorm:"size:100;column:asset_no" json:"asset_no"`
	AssetTag             string   `gorm:"size:100;column:asset_tag" json:"asset_tag"`
	UnitNumber           string   `gorm:"size:50;column:unit_number" json:"unit_number"`
	Brand                string   `gorm:"size:100;column:brand" json:"brand"`
	Model                string   `gorm:"size:100;column:model" json:"model"`
	Status               string   `gorm:"size:50;default:'ACTIVE';column:status" json:"status"`
	CreatedAt            time.Time `gorm:"autoCreateTime;column:created_at" json:"created_at"`
	UpdatedAt            time.Time `gorm:"autoUpdateTime;column:updated_at" json:"updated_at"`

	// Associations (Belongs To)
	Category EquipmentCategory `gorm:"foreignKey:CategoryID;constraint:OnUpdate:CASCADE,OnDelete:RESTRICT;" json:"category,omitempty"`
	Building *Building         `gorm:"foreignKey:BuildingCode;constraint:OnUpdate:CASCADE,OnDelete:SET NULL;" json:"building,omitempty"`

	// Associations (Has One Specialized Specs)
	SpecTransformer     *SpecTransformer     `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_transformer,omitempty"`
	SpecGenerator       *SpecGenerator       `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_generator,omitempty"`
	SpecElectricalPanel *SpecElectricalPanel `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_electrical_panel,omitempty"`
	SpecFireAlarm       *SpecFireAlarm       `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_fire_alarm,omitempty"`
	SpecPump            *SpecPump            `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_pump,omitempty"`
	SpecPaddleWheel     *SpecPaddleWheel     `gorm:"foreignKey:EquipmentID;references:EquipmentID" json:"spec_paddle_wheel,omitempty"`
}

func (Equipment) TableName() string {
	return "equipments"
}
